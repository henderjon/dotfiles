
## INSTALL
--------------------------------------------------------------------------------

- $ /bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"
- $ brew bundle dump --file=path/Brewfile
- $ brew bundle --file=path/Brewfile
- $ brew install \
	tmux \
	shellcheck \
	gnupg \
	gpg-suite

## FONTS
--------------------------------------------------------------------------------

$ brew tap homebrew/cask-fonts
$ brew install --cask ./resources/brew/font-amazon-ember.rb

## GETTING INSTALL INFO
--------------------------------------------------------------------------------

$ brew info $bin

## GETTING A LIST OF INSTALLED BREWS
--------------------------------------------------------------------------------

$ brew leaves

## UPDATE/UPGRADE
--------------------------------------------------------------------------------

update pulls the latest changes
upgrade install the latest changes
